Some usefull ovs debug command
#### DB
```shell
ovs-vsctl show

sudo ovs-vsctl get open . external-ids

# get all interface
ovs-vsctl list interface
## list specific interface
ovs-vsctl list interface vnet3

ovs-vsctl list open_vswitch
ovs-vsctl --columns=options list interface vnet3
ovs-vsctl --columns=ofport,name list interface
ovs-vsctl --columns=ofport,name --format=table list interface
ovs-vsctl -f csv --no-heading --columns=_uuid list controller
ovs-vsctl -f csv --no-heading -d bare --columns=other_config list port
ovs-vsctl --format=table --columns=name,mac_in_user find interface name=vnet3

ovs-vsctl set port vnet3 tag=1927
ovs-vsctl get port vnet3 tag
ovs-vsctl remote port net3 tag 1927

ovs-vsctl set interface vnet3 mac='5c:b9:01:00:00:01'

ovs-vsctl clear Bridge br0 stp_enable
ovs-vsctl --may-exist add-br br0 -- set bridge br0 datapath_type=netdev
ovs-vsctl --if-exists del-br br0



```

sudo ovn-nbctl show
